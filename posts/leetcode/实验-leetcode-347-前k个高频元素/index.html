Step 3 维护大小为 K 的小顶堆

class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            freqMap.put(num, freqMap.getOrDefault(num, 0) + 1);
        }

        PriorityQueue<Integer> minHeap = new PriorityQueue<>(
            (a, b) -> freqMap.get(a) - freqMap.get(b)
        );

        for (int num : freqMap.keySet()) {
            if (minHeap.size() < k) {
                minHeap.offer(num);
            } else {
                if (freqMap.get(num) > freqMap.get(minHeap.peak())) {
                    minHeap.poll();
                    minHeap.offer(num);
                }
            }
        }
    }
}

Step 4 将堆中的数组弹出

class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            freqMap.put(num, freqMap.getOrDefault(num, 0) + 1);
        }

        PriorityQueue<Integer> minHeap = new PriorityQueue<>(
            (a, b) -> freqMap.get(a) - freqMap.get(b)
        );

        for (int num : freqMap.keySet()) {
            if (minHeap.size() < k) {
                minHeap.offer(num);
            } else {
                if (freqMap.get(num) > freqMap.get(minHeap.peek())) {
                    minHeap.poll();
                    minHeap.offer(num);
                }
            }
        }

        int[] res = new int[k];
        for (int i = 0; i < k; i ++) {
            res[i] = minHeap.poll();
        }
        return res;
    }
}
